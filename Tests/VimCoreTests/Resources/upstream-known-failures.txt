# codemirror-vim tests that VimKit fails on purpose. One test name per line.

# Tag text objects work lexically, in any text (as in Vim). CodeMirror 6 only
# finds tags with an XML/HTML syntax tree, so it expects `dat` in JavaScript to do nothing.
vim_dat_noop

# Browser-only: <C-c> in visual mode lets the browser's copy event fire. On macOS
# copy is Cmd-C, and VimKit treats <C-c> like Vim does (as <Esc>).
vim_<C-c>copy

# :registers shows a register on one line, with newlines as ^J (as Vim does);
# codemirror-vim prints them as line breaks.
vim_yank_visual_block
vim_yank_append_line_to_line_register
vim_yank_append_line_to_word_register
vim_yank_append_word_to_line_register

# Characterwise P leaves the cursor on the last pasted character, as in Vim
# (checked by the Vim oracle); codemirror-vim puts it one past.
vim_P

# :s leaves the cursor at the first non-blank of the last changed line, as in
# Vim (checked by the Vim oracle); codemirror-vim leaves it at the last match.
vim_macro_last_ex_command_register

# J adds no space after a line that ends in white space, as in Vim (checked by
# the Vim oracle); codemirror-vim keeps it and adds one more.
vim_J
vim_J_repeat
vim_visual_join_blank

# S<Esc> (and cc<Esc>, o<Esc>) with nothing typed removes the indent, as in
# Vim with 'autoindent' (checked by the Vim oracle); codemirror-vim keeps it.
vim_S_normal

# & in a :s replacement is the whole match, as in Vim (checked by the Vim
# oracle); this test expects a literal "$&".
vim_ex_substitute_javascript

# i{ / i[ / i< over brackets on their own lines is linewise and keeps those
# lines, as in Vim (checked by the Vim oracle); codemirror-vim joins them.
vim_di{_middle_spc
vim_di}_middle_spc
vim_di[_middle_spc
vim_di]_middle_spc
vim_di<_middle_spc
vim_di>_middle_spc
vim_vi{_middle_spc
